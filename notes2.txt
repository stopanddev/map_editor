// Add these to your separated state structs (keeps appstate clean)
struct Camera {
    float x = 0.0f, y = 0.0f;  // World position (top-left of viewport)
    float zoom = 1.0f;
    float speed = 300.0f;       // Pixels per second
};

struct TileMap {
    SDL_Texture* atlas = nullptr;  // Your large PNG loaded as texture
    int tile_width = 32;
    int tile_height = 32;
    int atlas_cols = 16;          // Tiles per row in atlas (adjust to your PNG)
    // Map data: for simplicity, a 2D vector or flat array of tile IDs
    // (0 = empty; >0 = tile ID in atlas)
    std::vector<int> tiles;       // Flat: index = ty * map_width + tx
    int map_width_tiles = 0;
    int map_height_tiles = 0;
};

struct AppState {
    // ... existing (window, renderer, input, etc.)
    Camera* camera = nullptr;
    TileMap* tilemap = nullptr;
};

Step 1: Initialize in SDL_AppInit

// After creating renderer...
state->camera = new Camera();

// Load your large PNG atlas
SDL_Surface* surf = SDL_LoadBMP("your_tilemap.png");  // Or SDL_LoadPNG/SDL_LoadJPG
if (surf) {
    state->tilemap = new TileMap();
    state->tilemap->atlas = SDL_CreateTextureFromSurface(state->renderer, surf);
    SDL_DestroySurface(surf);
    
    // Query sizes (or hardcode)
    int atlas_w, atlas_h;
    SDL_QueryTexture(state->tilemap->atlas, nullptr, nullptr, &atlas_w, &atlas_h);
    state->tilemap->atlas_cols = atlas_w / state->tilemap->tile_width;
    
    // Load your map data (e.g., from file)
    // Example: 100x100 map, populate state->tilemap->tiles with IDs
    state->tilemap->map_width_tiles = 100;
    state->tilemap->map_height_tiles = 100;
    state->tilemap->tiles.resize(100*100, 1);  // All tile ID 1 for demo
}

    state->camera->x += (state->input->drag_start.x - mouse.x) / state->camera->zoom;
    state->camera->y += (state->input->drag_start.y - mouse.y) / state->camera->zoom;
    state->input->drag_start = mouse;
}

Step 3: Efficient Rendering (in SDL_AppIterate)
C++

SDL_AppResult SDL_AppIterate(void* appstate_) {
    AppState* state = static_cast<AppState*>(appstate_);
    
    // Clear screen
    SDL_SetRenderDrawColor(state->renderer, 100, 149, 237, 255);  // Sky blue bg
    SDL_RenderClear(state->renderer);
    
    RenderTileMap(state);
    
    SDL_RenderPresent(state->renderer);
    return SDL_APP_CONTINUE;
}

void RenderTileMap(AppState* state) {
    if (!state->tilemap->atlas) return;
    
    Camera* cam = state->camera;
    TileMap* map = state->tilemap;
    auto renderer = state->renderer;
    
    // Screen size
    int screen_w, screen_h;
    SDL_GetRendererOutputSize(renderer, &screen_w, &screen_h);
    
    // Visible world rect (in world pixels)
    SDL_FRect visible_world = {
        cam->x, cam->y,
        (float)screen_w / cam->zoom,
        (float)screen_h / cam->zoom
    };
    
    // Visible tile range (only render these!)
    int start_tx = std::max(0, (int)(visible_world.x / map->tile_width));
    int end_tx = std::min(map->map_width_tiles, (int)ceil((visible_world.x + visible_world.w) / map->tile_width) + 1);
    int start_ty = std::max(0, (int)(visible_world.y / map->tile_height));
    int end_ty = std::min(map->map_height_tiles, (int)ceil((visible_world.y + visible_world.h) / map->tile_height) + 1);
    
    // Render only visible tiles
    for (int ty = start_ty; ty < end_ty; ++ty) {
        for (int tx = start_tx; tx < end_tx; ++tx) {
            size_t idx = (size_t)ty * map->map_width_tiles + tx;
            int tile_id = map->tiles[idx];
            if (tile_id == 0) continue;  // Skip empty
            
            // Source rect: tile region in large PNG atlas
            SDL_FRect src = {
                (float)((tile_id % map->atlas_cols) * map->tile_width),
                (float)((tile_id / map->atlas_cols) * map->tile_height),
                (float)map->tile_width,
                (float)map->tile_height
            };
            
            // Destination: world pos → screen pos (offset by camera, scaled)
            float world_x = (float)tx * map->tile_width;
            float world_y = (float)ty * map->tile_height;
            SDL_FRect dst = {
                world_x - cam->x,
                world_y - cam->y,
                (float)map->tile_width * cam->zoom,
                (float)map->tile_height * cam->zoom
            };
            
            SDL_RenderTexture(renderer, map->atlas, &src, &dst);
        }
    }
}
